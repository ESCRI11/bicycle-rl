======================  COPY THIS SKELETON  ======================
Below is a COMPLETE, ALREADY-CORRECT bicycle. Reproduce its structural lines (section 1-4)
EXACTLY as written — same coordinates, same radii, same order, same vertex lists. Every
past failure came from a model "improvising" the wheels, frame, fork, or chain instead of
copying these numbers, which breaks the geometric checklist (equal wheels, closed frame,
steering, drivetrain). You MAY freely change the colour hex values, wash opacity/texture
numbers, add spokes/saddle/hatching, and add brush.wiggle(0.1-0.3) to non-critical lines —
but do NOT change any coordinate, radius, or vertex list in sections 1-4, and do NOT delete
any of those calls.

function paint() {
  const INK_COLOUR = '#2b2320';
  const WASH_A = '#8b3a3a';
  const WASH_B = '#3a5f8b';

  const R  = 95;
  const RH = [-120, 90];   // rear hub
  const FH = [ 120, 90];   // front hub
  const BB = [ -40, 90];   // bottom bracket
  const ST = [ -90, -60];  // top of seat tube
  const HT = [  95, -40];  // head tube top
  const HB = [  60, -90];  // handlebar grip

  // ---- 0. WASH (paint first, behind everything) ----
  brush.fill(WASH_A, 90);
  brush.fillBleed(0.15, 'out');
  brush.fillTexture(0.6, 0.4);
  brush.polygon([[RH[0]-R, -20], [FH[0]+R, -20], [FH[0]+R, 160], [RH[0]-R, 160]]);
  brush.noFill();

  // ---- 1. WHEELS (equal radius, perfectly circular — do not touch these lines) ----
  brush.noField();
  brush.set('rotring', INK_COLOUR, 2);
  brush.circle(RH[0], RH[1], R, false);
  brush.circle(FH[0], FH[1], R, false);
  brush.circle(RH[0], RH[1], 6, false);
  brush.circle(FH[0], FH[1], 6, false);
  for (let i = 0; i < 4; i++) {
    const a = i * (TWO_PI / 4);
    brush.line(RH[0], RH[1], RH[0] + R * cos(a), RH[1] + R * sin(a));
    brush.line(FH[0], FH[1], FH[0] + R * cos(a), FH[1] + R * sin(a));
  }

  // ---- 2. CLOSED FRAME (one continuous closed polygon — do not touch these vertices) ----
  brush.set('pen', INK_COLOUR, 3);
  brush.polygon([RH, BB, HT, ST, RH]);

  // ---- 3. FORK + STEERING (must touch FH and HT exactly — do not touch these coords) ----
  brush.line(HT[0], HT[1], FH[0], FH[1]);
  brush.line(HT[0], HT[1], HB[0], HB[1]);
  brush.line(HB[0] - 15, HB[1] - 8, HB[0] + 15, HB[1] + 8);

  // ---- 4. DRIVETRAIN (chainring at BB, chain to RH — do not touch these coords) ----
  brush.circle(BB[0], BB[1], 24, false);
  brush.line(BB[0] + 22, BB[1] - 4, RH[0] + 18, RH[1] - 4);
  brush.line(BB[0] + 22, BB[1] + 4, RH[0] + 18, RH[1] + 4);
  brush.line(BB[0], BB[1], BB[0] + 18, BB[1] + 18);

  // ---- 5. SEAT + extra decoration (free to customise) ----
  brush.set('2B', INK_COLOUR, 2);
  brush.wiggle(0.15);
  brush.line(ST[0] - 12, ST[1] - 6, ST[0] + 12, ST[1] - 6);
  brush.wiggle(0);

  brush.fill(WASH_B, 70);
  brush.circle(BB[0], BB[1], 30, false);
  brush.noFill();
}
